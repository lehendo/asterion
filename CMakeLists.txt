cmake_minimum_required(VERSION 3.24)
project(asterion VERSION 0.1.0 LANGUAGES CXX)
option(ASTERION_BUILD_NATIVE "Build stable host semantics library" OFF)
option(ASTERION_ENABLE_CUDA "Enable experimental CUDA native backend (not implemented)" OFF)
option(ASTERION_BUILD_TESTS "Build native host tests" ON)
option(ASTERION_SANITIZE "Enable address/undefined sanitizers" OFF)
if(ASTERION_ENABLE_CUDA)
  message(FATAL_ERROR "Native CUDA/MLIR lowering is not implemented; use generated Triton backend.")
endif()
if(ASTERION_BUILD_NATIVE)
  add_library(asterion_native SHARED cpp/src/core.cpp)
  target_include_directories(asterion_native PUBLIC cpp/include)
  target_compile_features(asterion_native PUBLIC cxx_std_20)
  target_compile_options(asterion_native PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(ASTERION_SANITIZE)
    target_compile_options(asterion_native PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(asterion_native PUBLIC -fsanitize=address,undefined)
  endif()
  if(ASTERION_BUILD_TESTS)
    include(FetchContent)
    FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git
      GIT_TAG 56809e5282f104c5c8b570e7c2996cdc352d94f1 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(Catch2)
    enable_testing()
    add_executable(asterion_native_tests cpp/tests/test_core.cpp)
    target_link_libraries(asterion_native_tests PRIVATE asterion_native Catch2::Catch2WithMain)
    target_compile_features(asterion_native_tests PRIVATE cxx_std_20)
    target_compile_options(asterion_native_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    include(Catch)
    catch_discover_tests(asterion_native_tests)
  endif()
endif()

option(ASTERION_BUILD_DRIVER_RUNTIME "Build optional CUDA Driver host launcher" OFF)
if(ASTERION_BUILD_DRIVER_RUNTIME)
  find_package(CUDAToolkit 12 REQUIRED)
  add_library(asterion_cuda_driver SHARED cpp/src/driver.cpp)
  target_include_directories(asterion_cuda_driver PUBLIC cpp/include)
  target_compile_features(asterion_cuda_driver PUBLIC cxx_std_20)
  target_compile_options(asterion_cuda_driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(asterion_cuda_driver PRIVATE CUDA::cuda_driver)
  # Toolkit stubs are link-time only; resolve the installed NVIDIA driver at runtime.
  set_target_properties(asterion_cuda_driver PROPERTIES SKIP_BUILD_RPATH TRUE)
endif()
